-- Referência legível da política raw usada por _dominio_num_if_produto.
-- V_FAMILIA_CONTAS, V_OBJETOS_SERVICO e V_PARAMETRO_SIC ficam no validador Python.
SELECT DISTINCT f.NUM_IF
FROM FLAGS f
    INNER JOIN CETIP.OPERACAO o
        ON o.NUM_IF = f.NUM_IF
    INNER JOIN CETIP.DADO_OPERACAO do
        ON do.NUM_ID_OPERACAO = o.NUM_ID_OPERACAO
    INNER JOIN CETIP.LANCAMENTO l
        ON l.NUM_ID_OPERACAO = o.NUM_ID_OPERACAO
    INNER JOIN CETIP.ESPECIFICACAO e2
        ON e2.NUM_ID_OPERACAO = o.NUM_ID_OPERACAO
    INNER JOIN CETIP.ESPECIFICACAO_COMITENTE ec
        ON ec.NUM_ID_ESPECIFICACAO = e2.NUM_ID_ESPECIFICACAO
WHERE NOT EXISTS (
    -- Uma operação inválida elimina o NUM_IF inteiro; uma operação válida não a mascara.
    SELECT 1
    FROM CETIP.OPERACAO ox
        LEFT JOIN CETIP.TIPO_OPER_OBJETO_SERV tos
            ON tos.NUM_ID_TIPO_OPER_OBJETO_SERV = ox.NUM_ID_TIPO_OPER_OBJETO_SERV
        LEFT JOIN CETIP.TIPO_OPERACAO top
            ON top.NUM_ID_TIPO_OPERACAO = tos.NUM_ID_TIPO_OPERACAO
        LEFT JOIN CETIP.CONTA_PARTICIPANTE p1
            ON p1.NUM_CONTA_PARTICIPANTE = ox.NUM_CONTA_PARTICIPANTE_P1
        LEFT JOIN CETIP.CONTA_PARTICIPANTE p2
            ON p2.NUM_CONTA_PARTICIPANTE = ox.NUM_CONTA_PARTICIPANTE_P2
    WHERE ox.NUM_IF = f.NUM_IF
      AND (
          TRIM(TO_CHAR(ox.NUM_ID_TIPO_OPER_OBJETO_SERV)) IS NULL
          OR tos.NUM_ID_TIPO_OPER_OBJETO_SERV IS NULL
          OR tos.NUM_ID_OBJETO_SERVICO <> 44
          OR tos.NUM_ID_OBJETO_SERVICO IS NULL
          OR TRIM(tos.IND_DISPONIVEL_IDENTIFICACAO) <> 'S'
          OR tos.IND_DISPONIVEL_IDENTIFICACAO IS NULL
          OR top.NUM_ID_TIPO_OPERACAO IS NULL
          OR top.COD_TIPO_OPERACAO <> 1
          OR top.COD_TIPO_OPERACAO IS NULL
          OR TRIM(TO_CHAR(ox.NUM_CONTA_PARTICIPANTE_P1)) IS NULL
          OR p1.NUM_CONTA_PARTICIPANTE IS NULL
          OR p1.NUM_ID_SITUACAO_CONTA <> 1
          OR p1.NUM_ID_SITUACAO_CONTA IS NULL
          OR TRIM(p1.COD_CONTA_PARTICIPANTE) IS NULL
          OR NOT REGEXP_LIKE(TRIM(p1.COD_CONTA_PARTICIPANTE), '^[0-9]{5}\.(40|10)-[0-9]$')
          OR TRIM(TO_CHAR(ox.NUM_CONTA_PARTICIPANTE_P2)) IS NULL
          OR p2.NUM_CONTA_PARTICIPANTE IS NULL
          OR p2.NUM_ID_SITUACAO_CONTA <> 1
          OR p2.NUM_ID_SITUACAO_CONTA IS NULL
          OR TRIM(p2.COD_CONTA_PARTICIPANTE) IS NULL
          OR NOT REGEXP_LIKE(TRIM(p2.COD_CONTA_PARTICIPANTE), '^[0-9]{5}\.(40|10)-[0-9]$')
      )
)
AND NOT EXISTS (
    SELECT 1
    FROM CETIP.TITULO t
        LEFT JOIN CETIP.CONTA_PARTICIPANTE cp
            ON cp.NUM_CONTA_PARTICIPANTE = t.NUM_CONTA_PARTICIPANTE
    WHERE t.NUM_IF = f.NUM_IF
      AND t.COD_TIPO_ESCALONAMENTO IS NULL
      AND t.NUM_CONTA_PARTICIPANTE IS NOT NULL
      AND (
          cp.NUM_CONTA_PARTICIPANTE IS NULL
          OR cp.NUM_ID_SITUACAO_CONTA <> 1
          OR cp.NUM_ID_SITUACAO_CONTA IS NULL
          OR TRIM(cp.COD_CONTA_PARTICIPANTE) IS NULL
          OR NOT REGEXP_LIKE(TRIM(cp.COD_CONTA_PARTICIPANTE), '^[0-9]{5}\.(40|10)-[0-9]$')
      )
)
AND NOT EXISTS (
    SELECT 1
    FROM CETIP.DEPOSITO_AUTOMATICO_IF d
        LEFT JOIN CETIP.CONTA_PARTICIPANTE cp
            ON cp.NUM_CONTA_PARTICIPANTE = d.NUM_CONTA_PARTICIPANTE
    WHERE d.NUM_IF = f.NUM_IF
      AND d.NUM_CONTA_PARTICIPANTE IS NOT NULL
      AND (
          cp.NUM_CONTA_PARTICIPANTE IS NULL
          OR cp.NUM_ID_SITUACAO_CONTA <> 1
          OR cp.NUM_ID_SITUACAO_CONTA IS NULL
          OR TRIM(cp.COD_CONTA_PARTICIPANTE) IS NULL
          OR NOT REGEXP_LIKE(TRIM(cp.COD_CONTA_PARTICIPANTE), '^[0-9]{5}\.(40|10)-[0-9]$')
      )
);
