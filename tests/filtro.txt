SELECT DISTINCT f.NUM_IF FROM FLAGS f
    INNER JOIN CETIP.OPERACAO o
        ON f.num_if = o.NUM_IF
    INNER JOIN CETIP.DADO_OPERACAO do
        ON do.NUM_ID_OPERACAO = o.NUM_ID_OPERACAO
    INNER JOIN CETIP.LANCAMENTO l
        ON l.NUM_ID_OPERACAO = o.NUM_ID_OPERACAO
    INNER JOIN CETIP.ESPECIFICACAO e2
        ON e2.NUM_ID_OPERACAO = o.NUM_ID_OPERACAO
    INNER JOIN CETIP.ESPECIFICACAO_COMITENTE ec
        ON ec.NUM_ID_ESPECIFICACAO = e2.NUM_ID_ESPECIFICACAO
    -- >>> ponte para chegar em COD_TIPO_OPERACAO
    INNER JOIN CETIP.TIPO_OPER_OBJETO_SERV toos
        ON toos.NUM_ID_TIPO_OPER_OBJETO_SERV = o.NUM_ID_TIPO_OPER_OBJETO_SERV
    INNER JOIN CETIP.TIPO_OPERACAO top
        ON top.NUM_ID_TIPO_OPERACAO = toos.NUM_ID_TIPO_OPERACAO
    -- >>> conta da ponta P1 para chegar em NUM_ID_SITUACAO_CONTA
    INNER JOIN CETIP.CONTA_PARTICIPANTE cp
        ON cp.NUM_CONTA_PARTICIPANTE = o.NUM_CONTA_PARTICIPANTE_P1
WHERE top.COD_TIPO_OPERACAO = 1
  AND cp.NUM_ID_SITUACAO_CONTA IN (1, 2);
